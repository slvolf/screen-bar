以下是基于 **CMSIS框架** 在Win11 + VSCode + PlatformIO环境下开发CH32F103C8T6的完整流程（寄存器级操作，利用CMSIS的标准化内核接口简化开发）：


### **一、环境准备**  
#### 1. 基础软件安装  
- 安装 [VSCode](https://code.visualstudio.com/)（Win11版本）；  
- 打开VSCode，在扩展商店搜索 **PlatformIO IDE** 并安装（建议更新至最新版本，确保框架选项完整）。  

#### 2. 安装STSTM32平台及CMSIS依赖  
CMSIS是ARM官方的内核接口标准，STSTM32平台已内置兼容Cortex-M3的CMSIS库：  
- 打开PlatformIO主页（左侧“外星人”图标）；  
- 进入 **Platforms > Embedded**，搜索 **ST STM32** 并安装（包含ARM GCC工具链、CMSIS核心库）；  
- 命令安装：`pio pkg install --global --platform ststm32`（自动包含CMSIS组件）。  


### **二、新建项目（选择CMSIS框架）**  
1. 在PlatformIO主页点击 **New Project**；  
2. 项目配置：  
   - **Name**：自定义名称（如 `ch32f103_cmsis`）；  
   - **Board**：搜索并选择 `genericSTM32F103C8T6`（与CH32F103硬件参数完全兼容）；  
   - **Framework**：选择 **CMSIS**（标准化内核接口，支持寄存器操作）；  
   - **Location**：选择项目保存路径（如 `D:\Projects`）；  
3. 点击 **Finish**，PlatformIO会自动生成包含CMSIS库的项目结构（核心文件在 `lib/CMSIS` 目录）。  


### **三、项目配置（`platformio.ini`）**  
打开项目根目录的 `platformio.ini`，替换为以下配置（适配CH32F103C8T6）：  
```ini  
[env:genericSTM32F103C8T6]  
platform = ststm32  ; 基于STM32平台（兼容CH32）  
board = genericSTM32F103C8T6  ; 兼容板型  
framework = cmsis  ; 使用CMSIS框架（内核接口）  
toolchain = gcc-arm-none-eabi  ; ARM GCC编译器  

; 上传配置（二选一）  
; 方案1：WCH-Link调试器（推荐，支持下载+调试）  
upload_protocol = wch-link  
debug_tool = wch-link  
debug_init_break = tbreak main  ; 调试时在main处断住  

; 方案2：USB转串口上传  
; upload_protocol = serial  
; upload_port = COM3  ; 替换为实际串口（设备管理器查看）  
; upload_speed = 115200  

; 编译参数（适配CH32F103C8T6）  
build_flags =  
    -mcpu=cortex-m3  ; 明确Cortex-M3内核  
    -mthumb          ; 启用Thumb指令集  
    -DCH32F103C8T6   ; 自定义宏，区分芯片型号  
    -Iinclude        ; 包含自定义头文件目录（若有）  

; 链接脚本（指定CH32内存布局）  
board_build.ldscript = ch32f103c8t6.ld  
```  


### **四、编写链接脚本（`ch32f103c8t6.ld`）**  
在项目根目录创建该文件，定义CH32F103C8T6的内存布局（与STM32F103一致）：  
```ld  
/* CH32F103C8T6：64KB Flash，20KB RAM */  
MEMORY  
{  
    FLASH (rx)  : ORIGIN = 0x08000000, LENGTH = 64K  
    RAM   (xrw) : ORIGIN = 0x20000000, LENGTH = 20K  
}  

ENTRY(Reset_Handler)  /* 程序入口：复位向量 */  

SECTIONS  
{  
    /* 中断向量表（CMSIS要求放在Flash起始位置） */  
    .isr_vector : { KEEP(*(.isr_vector)) } >FLASH  

    /* 代码段（包含CMSIS内核代码和用户程序） */  
    .text : {  
        *(.text)           /* 用户代码 */  
        *(.text*)          /* 扩展代码 */  
        *(.rodata)         /* 只读数据 */  
        *(.rodata*)        /* 扩展只读数据 */  
    } >FLASH  

    /* 初始化数据段（从Flash复制到RAM） */  
    .data : AT(LOADADDR(.text) + SIZEOF(.text))  
    {  
        _sdata = .;  
        *(.data)  
        _edata = .;  
    } >RAM  

    /* 未初始化数据段（BSS） */  
    .bss : {  
        _sbss = .;  
        *(.bss)  
        _ebss = .;  
    } >RAM  

    /* 栈顶地址（RAM末尾，CMSIS依赖此定义） */  
    _estack = ORIGIN(RAM) + LENGTH(RAM);  
}  
```  


### **五、编写启动文件（`src/startup_ch32f103.s`）**  
CMSIS依赖标准化的启动流程，在`src`目录创建启动文件（兼容CMSIS的中断向量表和初始化逻辑）：  
```s  
    .syntax unified  
    .cpu cortex-m3  /* 适配Cortex-M3（CMSIS要求） */  
    .thumb  

    /* 中断向量表（CMSIS格式，可扩展其他中断） */  
    .section .isr_vector, "a", %progbits  
    .align 2  
    .long _estack          /* 栈顶地址（来自链接脚本） */  
    .long Reset_Handler    /* 复位处理函数 */  
    .long NMI_Handler      /* NMI中断（CMSIS标准） */  
    .long HardFault_Handler /* 硬件错误中断 */  
    /* 其他中断可在此扩展（如SysTick、USART等） */  

    /* 复位处理函数：初始化系统并跳转至main（兼容CMSIS） */  
    .section .text.Reset_Handler, "ax", %progbits  
    .align 2  
Reset_Handler:  
    /* 1. 复制.data段到RAM（CMSIS要求的初始化） */  
    ldr r0, =_sdata  
    ldr r1, =_edata  
    ldr r2, =_sidata  
copy_data:  
    cmp r0, r1  
    it lt  
    ldrlt r3, [r2], #4  
    strlt r3, [r0], #4  
    blt copy_data  

    /* 2. 清零.bss段 */  
    ldr r0, =_sbss  
    ldr r1, =_ebss  
    mov r2, #0  
zero_bss:  
    cmp r0, r1  
    it lt  
    strlt r2, [r0], #4  
    blt zero_bss  

    /* 3. 调用CMSIS系统初始化（可选，用于时钟配置等） */  
    bl SystemInit  

    /* 4. 跳转到main函数 */  
    bl main  
    b .  

    /* 空中断处理函数（CMSIS占位） */  
    .weak NMI_Handler  
    NMI_Handler:  
    b .  

    .weak HardFault_Handler  
    HardFault_Handler:  
    b .  

    /* 声明链接脚本符号（供CMSIS使用） */  
    .extern _sidata  
    .extern _sdata  
    .extern _edata  
    .extern _sbss  
    .extern _ebss  
    .extern _estack  
    .extern SystemInit  /* CMSIS系统初始化函数（用户实现） */  

    .end  
```  


### **六、编写系统初始化文件（`src/system_ch32f103.c`）**  
CMSIS要求实现`SystemInit`函数（初始化时钟等），在`src`目录创建：  
```c  
#include "stm32f1xx.h"  /* CMSIS通用头文件（兼容CH32） */  

/* 系统时钟初始化（CH32F103默认使用内部8MHz HSI，此处简化配置） */  
void SystemInit(void) {  
    // 使能外设时钟（如需配置外部晶振，可在此扩展）  
    RCC->APB2ENR |= RCC_APB2ENR_AFIOEN;  // 使能复用功能时钟（可选）  
}  
```  


### **七、编写主程序（`src/main.c`）**  
使用CMSIS的外设寄存器定义（结构体形式，更规范）实现点灯：  
```c  
#include "stm32f1xx.h"  /* CMSIS提供的外设寄存器结构体（兼容CH32） */  
#include <stdint.h>  

/* 延时函数（约1ms） */  
void delay_ms(uint32_t ms) {  
    for (volatile uint32_t i = 0; i < ms * 8000; i++);  
}  

int main(void) {  
    /* 1. 使能GPIOC时钟（使用CMSIS的RCC结构体） */  
    RCC->APB2ENR |= RCC_APB2ENR_IOPCEN;  

    /* 2. 配置PC13为推挽输出（50MHz）  
       - GPIOx_CRH：端口配置高寄存器（引脚8-15）  
       - PC13对应bit20-23：MODE13=10（50MHz），CNF13=00（推挽输出）  
    */  
    GPIOC->CRH &= ~(GPIO_CRH_MODE13 | GPIO_CRH_CNF13);  // 清除原配置  
    GPIOC->CRH |= GPIO_CRH_MODE13_1;  // MODE13=10（50MHz）  
    // CNF13默认00（推挽输出），无需额外配置  

    /* 3. 循环翻转PC13电平（LED闪烁） */  
    while (1) {  
        GPIOC->ODR ^= GPIO_ODR_ODR13;  // 翻转PC13（使用CMSIS的位定义）  
        delay_ms(500);  
    }  
}  
```  


### **八、编译与上传**  
#### 1. 编译程序  
- 点击VSCode底部状态栏的 **✓ 图标**（Build）；  
- 编译成功会显示 `SUCCESS`，生成文件在 `/.pio/build/genericSTM32F103C8T6` 目录。  

#### 2. 硬件连接与上传  
##### 方案1：WCH-Link调试器  
- **接线**：SWCLK接CLK，SWDIO接DIO，GND接GND，3.3V接VCC；  
- **上传**：点击底部 **→ 图标**，成功后LED闪烁。  

##### 方案2：USB转串口上传  
- **接线**：TX接PA10，RX接PA9，GND接GND，3.3V接VCC；  
- **BOOT配置**：烧录时`BOOT0=高`、`BOOT1=低`，按复位键后点击上传；  
- 上传完成后`BOOT0=低`，按复位键启动程序。  


### **九、调试（WCH-Link支持）**  
1. 点击底部 **🐞 图标**（Debug）；  
2. 支持断点、单步执行（F10）、监视寄存器（如`GPIOC->ODR`）等功能，调试入口在`main`。  


### **关键优势与注意事项**  
#### 核心优势  
- **CMSIS标准化**：使用`stm32f1xx.h`提供的外设结构体（如`RCC->APB2ENR`），比直接写寄存器地址更规范，兼容CH32的寄存器布局；  
- **内核接口支持**：可直接调用CMSIS的内核函数（如`NVIC_EnableIRQ`、`SysTick_Config`），方便扩展中断和定时器功能；  
- **兼容性强**：CH32与STM32的CMSIS接口完全兼容，无需修改库文件即可使用。  

#### 常见问题  
- **“找不到stm32f1xx.h”**：确保STSTM32平台安装完整，可通过`pio pkg update`更新平台；  
- **编译报错“undefined reference to SystemInit”**：检查`system_ch32f103.c`是否在`src`目录，且实现了`SystemInit`函数；  
- **调试无反应**：确认WCH-Link驱动已安装，且SWD接线正确（CLK/DIO未接反）。  


通过以上流程，可借助CMSIS框架的标准化接口简化CH32F103C8T6的开发，同时保留寄存器级操作的灵活性，兼顾规范性和可控性。
